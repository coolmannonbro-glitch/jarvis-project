// Функція для оновлення часу та дати function updateDateTime() { const now = new Date(); const day = String(now.getDate()).padStart(2, '0'); const month = String(now.getMonth() + 1).padStart(2, '0'); const year = now.getFullYear(); document.getElementById('widget-date').innerText = `DATE: ${day}/${month}/${year}`; const hours = String(now.getHours()).padStart(2, '0'); const minutes = String(now.getMinutes()).padStart(2, '0'); const seconds = String(now.getSeconds()).padStart(2, '0'); document.getElementById('widget-time').innerText = `TIME: ${hours}:${minutes}:${seconds}`; } setInterval(updateDateTime, 1000); updateDateTime(); const inputField = document.getElementById('user-command'); inputField.addEventListener('keypress', function(e) { if (e.key === 'Enter') sendCommand(); }); function printToConsole(sender, text, isJarvis = false) { const consoleOutput = document.getElementById('console-output'); const line = document.createElement('div'); line.className = 'console-line'; if (isJarvis) { line.innerHTML = `${sender}: ${text}`; } else { line.innerHTML = `${sender}: ${text}`; } consoleOutput.appendChild(line); consoleOutput.scrollTop = consoleOutput.scrollHeight; } function sendCommand() { const cmd = inputField.value.trim().toLowerCase(); if (!cmd) return; printToConsole('Ви', inputField.value, false); inputField.value = ''; setTimeout(() => { if (cmd === 'help') { printToConsole('Джарвіс', 'Доступні команди: status, clear, protocol, hello, youtube', true); } else if (cmd === 'status') { printToConsole('Джарвіс', 'Всі модулі функціонують нормально. Температура ядра 38°C.', true); } else if (cmd === 'clear') { document.getElementById('console-output').innerHTML = ''; } else if (cmd === 'hello' || cmd === 'привіт') { printToConsole('Джарвіс', 'Привіт, сер! Завжди до ваших послуг.', true); } else if (cmd === 'protocol') { printToConsole('Джарвіс', 'Протокол захисту Робочого столу активовано. Слава Старку!', true); } else if (cmd === 'youtube' || cmd === 'ютуб' || cmd === 'відкрий ютуб') { printToConsole('Джарвіс', 'Відкриваю YouTube, сер. Приємного перегляду!', true); window.open('https://youtube.com', '_blank'); } else { printToConsole('Джарвіс', 'Запит обробляється, але у мене поки немає інструкцій для цієї команди. Спробуйте "help".', true); } }, 400); }
